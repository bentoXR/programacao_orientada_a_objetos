    }//construtor
    public Carro() {
    }//construtor

        return velocidade;
    }
    public void setModelo(String modelo) {
        this.modelo = modelo;
    }
    public void setVelocidade(int velocidade) {
        this.velocidade = velocidade;
    }
    @Override
    public String toString() {
        return "Carro{" +
                "modelo='" + modelo + '\'' +
                ", velocidade=" + velocidade +
                '}';
    }
    public void acelerar(){
        velocidade += 10;
        System.out.println("A velocidade aumentou em 10km/h. Velocidade atual: " + velocidade + "km/h.");
    }
    public void frear(){
        if(velocidade>= 10){
            velocidade -= 10;
            System.out.println("A velocidade diminuiu em 10km/h. Velocidade atual: " + velocidade + "km/h.");
        }
        else {
            System.out.println("Você já está parado.");
        }
    }
}

public class Livro {
    private String titulo;
    private String autor;
    private boolean emprestado;
    private int quantidadePaginas;
    private double preco; 
    public Livro(String titulo, String autor, boolean emprestado, int quantidadePaginas, double preco) {
        this.titulo = titulo;
        this.autor = autor;
        this.emprestado = emprestado;
        this.quantidadePaginas = quantidadePaginas;
        this.preco = preco;
    }
    public String getTitulo() {
        return titulo;
    }
    public String getAutor() {
        return autor;
    }
    public double getPreco() {
        return preco;
    }
    public void emprestar() {
        if (!emprestado) {
            emprestado = true;
            System.out.println("Livro emprestado com sucesso.");
        } else {
            System.out.println("O livro já está emprestado.");
        }
    }
    public void devolver() {
        if (emprestado) {
            emprestado = false;
            System.out.println("Livro devolvido com sucesso.");
        } else {
            System.out.println("O livro já está disponível.");
        }
    }
    public void exibirInformacoes() {
        System.out.println("Título: " + titulo);
        System.out.println("Autor: " + autor);
        System.out.println("Preço: R$ " + preco);
        System.out.println("Status: " + (emprestado ? "Emprestado" : "Disponível"));
    }
}

import java.util.ArrayList;
import java.util.List;
public class Biblioteca {
    private String nome;
    private List<Livro> livros;
    public Biblioteca(String nome) {
        this.nome = nome;
        this.livros = new ArrayList<>();
    }
    public void adicionarLivro(Livro livro) {
        livros.add(livro);
    }
    // Imprime o valor mais caro
    public void imprimirLivroMaisCaro() {
        double caro = 0;
        int pos = 0;
        if (!livros.isEmpty()) {
            for (int i = 0;i<livros.size();i++) {
                if (livros.get(i).getPreco() > caro) {
                    caro = livros.get(i).getPreco();
                    pos = i;
                }
            }
            System.out.println("valor mais caro: "+caro);
        }
    }
    // Imprime o nome do livro mais barato
    public void imprimirLivroMaisBarato() {
        if (!livros.isEmpty()) {
            Livro maisBarato = livros.get(0);
            for (Livro livro : livros) {
                if (livro.getPreco() < maisBarato.getPreco()) {
                    maisBarato = livro;
                }
            }
            System.out.println("Livro mais barato: " + maisBarato.getTitulo() + " (R$ " + maisBarato.getPreco() + ")");
        }
    }
    // Imprime a média de preço dos livros
    public void imprimirMediaPrecos() {
        if (livros.isEmpty()) {
            System.out.println("A biblioteca não possui livros cadastrados.");
            return;
        }
        double soma = 0;
        for (Livro livro : livros) {
            soma += livro.getPreco();
        }
        double media = soma / livros.size();
        System.out.println("Média de preço dos livros: R$ " + media);
    }
    // Informa quantos livros um determinado autor escreveu
    public void contarLivrosPorAutor(String nomeAutor) {
        int contador = 0;
        for (Livro livro : livros) {
            if (livro.getAutor().equalsIgnoreCase(nomeAutor)) {
                contador++;
            }
        }
        System.out.println("O autor " + nomeAutor + " possui " + contador + " livro(s) cadastrado(s).");
    }
}

public class ContaBancaria {
    private String numeroConta;
    private double saldo;
    private String emailTitular;
    public ContaBancaria(String numeroConta, double saldoInicial, String emailTitular) {
        this.numeroConta = numeroConta;
        this.saldo = saldoInicial;
        this.emailTitular = emailTitular;
    }
    public void depositar(double valor) {
        if (valor > 0) {
            saldo += valor;
            System.out.println("Depósito de R$ " + valor +
                    " realizado na conta " + numeroConta);
        }
    }
    public void sacar(double valor, ServicoNotificacao notificacao) {
        if (valor > 0 && saldo >= valor) {
            saldo -= valor;
            System.out.println("Saque de R$ " + valor + " realizado com sucesso.");
            notificacao.enviarEmail(emailTitular, "Foi realizado um saque de R$ " + valor
                    + ". Saldo atual: R$ " + saldo);
        } else {
            System.out.println("Saldo insuficiente ou valor inválido para saque.");
        }
    }
    public double getSaldo() { return saldo; }
}
